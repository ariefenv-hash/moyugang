     1→     1→import {
     2→     2→  UnitInstance,
     3→     3→  HeroInstance,
     4→     4→  Projectile,
     5→     5→  Particle,
     6→     6→  DamageNumber,
     7→     7→  LevelConfig,
     8→     8→  InkLotusTrap,
     9→     9→  SealStamp,
    10→    10→  InkDragon,
    11→    11→  InkTreasure,
    12→    12→} from '../types/game';
    13→    13→
    14→    14→export class CanvasRenderer {
    15→    15→  private ctx: CanvasRenderingContext2D;
    16→    16→  private width: number;
    17→    17→  private height: number;
    18→    18→  /** 设备像素比：高分屏逻辑坐标 → 物理像素映射，画面更清晰 */
    19→    19→  private dpr: number = 1;
    20→    20→  private bgImage: HTMLImageElement | null = null;
    21→    21→
    22→    22→  constructor(ctx: CanvasRenderingContext2D, width: number, height: number, bgImageSrc?: string) {
    23→    23→    this.ctx = ctx;
    24→    24→    this.width = width;
    25→    25→    this.height = height;
    26→    26→
    27→    27→    if (bgImageSrc) {
    28→    28→      const img = new Image();
    29→    29→      img.src = bgImageSrc;
    30→    30→      img.onload = () => {
    31→    31→        this.bgImage = img;
    32→    32→      };
    33→    33→    }
    34→    34→  }
    35→    35→
    36→    36→  public resize(width: number, height: number, dpr?: number) {
    37→    37→    this.width = width;
    38→    38→    this.height = height;
    39→    39→    if (dpr && dpr > 0) this.dpr = dpr;
    40→    40→  }
    41→    41→
    42→    42→  // 2.5D Coordinate to Screen position with perspective
    43→    43→  private getDepthScale(laneY: number, laneMin: number, laneMax: number): number {
    44→    44→    const t = Math.max(0, Math.min(1, (laneY - laneMin) / (laneMax - laneMin)));
    45→    45→    return 0.86 + t * 0.22; // Depth scale 0.86 (back) to 1.08 (front)
    46→    46→  }
    47→    47→
    48→    48→  // Clear and render entire 2.5D scene with screen shake
    49→    49→  public render(
    50→    50→    cameraX: number,
    51→    51→    level: LevelConfig,
    52→    52→    worldWidth: number,
    53→    53→    laneMinY: number,
    54→    54→    laneMaxY: number,
    55→    55→    playerBaseHp: number,
    56→    56→    playerBaseMaxHp: number,
    57→    57→    enemyBaseHp: number,
    58→    58→    enemyBaseMaxHp: number,
    59→    59→    hero: HeroInstance,
    60→    60→    units: UnitInstance[],
    61→    61→    lotusTraps: InkLotusTrap[],
    62→    62→    projectiles: Projectile[],
    63→    63→    particles: Particle[],
    64→    64→    damageNumbers: DamageNumber[],
    65→    65→    sealStamps: SealStamp[],
    66→    66→    wells: { x: number; y: number; inkRemaining: number }[],
    67→    67→    gameTime: number,
    68→    68→    shakeX: number = 0,
    69→    69→    shakeY: number = 0,
    70→    70→    inkDragons: InkDragon[] = [],
    71→    71→    treasures: InkTreasure[] = [],
    72→    72→    inkSlowProgress: number = 0
    73→    73→  ) {
    74→    74→    const ctx = this.ctx;
    75→    75→    // 每帧重置基准变换到逻辑坐标系（物理分辨率由 dpr 缩放）
    76→    76→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
    77→    77→    ctx.save();
    78→    78→    ctx.translate(shakeX, shakeY);
    79→    79→    ctx.clearRect(0, 0, this.width, this.height);
    80→    80→
    81→    81→    // 1. Sky & Parallax Chinese Mountains
    82→    82→    this.drawSkyAndParallax(cameraX, level, laneMinY, gameTime);
    83→    83→
    84→    84→    // 2. 2.5D Perspective Ground Plane (Courtyard / Riverbank flagstones)
    85→    85→    this.drawPerspectiveGround(cameraX, worldWidth, laneMinY, laneMaxY, level, gameTime);
    86→    86→
    87→    87→    // 3. Ground Lotus Traps / Runes
    88→    88→    lotusTraps.forEach((trap) => {
    89→    89→      this.drawInkLotus(trap, cameraX, laneMinY, laneMaxY, gameTime);
    90→    90→    });
    91→    91→
    92→    92→    // 3.5 文房四宝掉落物（地面发光物）
    93→    93→    treasures.forEach((t) => {
    94→    94→      this.drawInkTreasure(t, cameraX, gameTime);
    95→    95→    });
    96→    96→
    97→    97→    // 4. Ink Wells
    98→    98→    wells.forEach((well) => {
    99→    99→      this.drawInkWell(well.x - cameraX, well.y, laneMinY, laneMaxY, well.inkRemaining, gameTime);
   100→   100→    });
   101→   101→
   102→   102→    // 5. Bases / Citadels
   103→   103→    const baseMidY = (laneMinY + laneMaxY) * 0.5;
   104→   104→    this.drawPlayerCitadel(100 - cameraX, baseMidY, playerBaseHp, playerBaseMaxHp, gameTime);
   105→   105→    this.drawEnemyCitadel(
   106→   106→      worldWidth - 100 - cameraX,
   107→   107→      baseMidY,
   108→   108→      enemyBaseHp,
   109→   109→      enemyBaseMaxHp,
   110→   110→      gameTime,
   111→   111→      level.boss?.name
   112→   112→    );
   113→   113→
   114→   114→    // 6. Draw Hero Inspiring Aura on Ground (Under feet)
   115→   115→    if (hero.state !== 'DEAD') {
   116→   116→      this.drawHeroAura25D(hero, cameraX, laneMinY, laneMaxY, gameTime);
   117→   117→    }
   118→   118→
   119→   119→    // 7. 2.5D Y-Depth Sorting for Characters and Drops
   120→   120→    type RenderItem =
   121→   121→      | { type: 'HERO'; y: number; data: HeroInstance }
   122→   122→      | { type: 'UNIT'; y: number; data: UnitInstance };
   123→   123→
   124→   124→    const renderList: RenderItem[] = [];
   125→   125→    if (hero.state !== 'DEAD') {
   126→   126→      renderList.push({ type: 'HERO', y: hero.y, data: hero });
   127→   127→    }
   128→   128→    units.forEach((u) => {
   129→   129→      renderList.push({ type: 'UNIT', y: u.y, data: u });
   130→   130→    });
   131→   131→
   132→   132→    renderList.sort((a, b) => a.y - b.y);
   133→   133→
   134→   134→    // Render sorted entities with skeletal animation
   135→   135→    renderList.forEach((item) => {
   136→   136→      if (item.type === 'HERO') {
   137→   137→        this.drawSkeletalHero(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   138→   138→      } else {
   139→   139→        this.drawSkeletalUnit(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   140→   140→      }
   141→   141→    });
   142→   142→
   143→   143→    // 7.5 墨龙卷（高连击奖励，飞在最上层）
   144→   144→    inkDragons.forEach((dragon) => {
   145→   145→      this.drawInkDragon(dragon, cameraX, laneMinY, laneMaxY, gameTime);
   146→   146→    });
   147→   147→
   148→   148→    // 8. Projectiles in 2.5D space
   149→   149→    projectiles.forEach((proj) => {
   150→   150→      this.drawProjectile(proj, cameraX, laneMinY, laneMaxY);
   151→   151→    });
   152→   152→
   153→   153→    // 9. Particles
   154→   154→    particles.forEach((p) => {
   155→   155→      this.drawParticle(p, cameraX);
   156→   156→    });
   157→   157→
   158→   158→    // 10. Seal Stamps (Red Calligraphy Stamps)
   159→   159→    sealStamps.forEach((stamp) => {
   160→   160→      this.drawSealStamp(stamp, cameraX);
   161→   161→    });
   162→   162→
   163→   163→    // 11. Floating Damage Numbers
   164→   164→    damageNumbers.forEach((d) => {
   165→   165→      this.drawDamageNumber(d, cameraX);
   166→   166→    });
   167→   167→
   168→   168→    ctx.restore();
   169→   169→
   170→   170→    // 12. 墨滞时刻全屏滤镜（屏幕空间，不受镜头/震动影响）
   171→   171→    if (inkSlowProgress > 0) {
   172→   172→      this.drawInkSlowVignette(inkSlowProgress, gameTime);
   173→   173→    }
   174→   174→  }
   175→   175→
   176→   176→  /** 墨龙卷：蛇形大笔触游走的墨龙 */
   177→   177→  private drawInkDragon(
   178→   178→    dragon: InkDragon,
   179→   179→    cameraX: number,
   180→   180→    laneMinY: number,
   181→   181→    laneMaxY: number,
   182→   182→    gameTime: number
   183→   183→  ) {
   184→   184→    const screenX = dragon.x - cameraX;
   185→   185→    if (screenX < -260 || screenX > this.width + 260) return;
   186→   186→
   187→   187→    const ctx = this.ctx;
   188→   188→    const scale = this.getDepthScale(dragon.y, laneMinY, laneMaxY);
   189→   189→    const t = gameTime * 9;
   190→   190→    const dir = dragon.vx >= 0 ? 1 : -1;
   191→   191→
   192→   192→    ctx.save();
   193→   193→    ctx.translate(screenX, dragon.y - 46 * scale);
   194→   194→
   195→   195→    // 龙身：多段蛇形波浪笔触（越靠尾部越细）
   196→   196→    const segs = 9;
   197→   197→    for (let i = 0; i < segs; i++) {
   198→   198→      const tt = t - i * 0.55;
   199→   199→      const sx = -dir * i * 26 * scale;
   200→   200→      const sy = Math.sin(tt) * 14 * scale;
   201→   201→      const segSize = (20 - i * 1.7) * scale;
   202→   202→      if (segSize <= 1) continue;
   203→   203→
   204→   204→      ctx.save();
   205→   205→      ctx.translate(sx, sy);
   206→   206→      ctx.fillStyle = `rgba(9, 9, 11, ${0.92 - i * 0.07})`;
   207→   207→      ctx.beginPath();
   208→   208→      ctx.ellipse(0, 0, segSize * 1.5, segSize * 0.85, Math.sin(tt) * 0.3, 0, Math.PI * 2);
   209→   209→      ctx.fill();
   210→   210→      // 金墨描边灵光
   211→   211→      ctx.strokeStyle = `rgba(250, 204, 21, ${0.5 - i * 0.05})`;
   212→   212→      ctx.lineWidth = 1.6;
   213→   213→      ctx.stroke();
   214→   214→      ctx.restore();
   215→   215→    }
   216→   216→
   217→   217→    // 龙首：张口吐墨 + 龙角 + 龙须
   218→   218→    const headSway = Math.sin(t) * 6 * scale;
   219→   219→    ctx.save();
   220→   220→    ctx.translate(dir * 10 * scale, headSway);
   221→   221→    ctx.scale(dir, 1);
   222→   222→
   223→   223→    // 头颅
   224→   224→    ctx.fillStyle = '#09090b';
   225→   225→    ctx.beginPath();
   226→   226→    ctx.ellipse(6 * scale, 0, 15 * scale, 10 * scale, 0, 0, Math.PI * 2);
   227→   227→    ctx.fill();
   228→   228→    // 吻部
   229→   229→    ctx.beginPath();
   230→   230→    ctx.moveTo(12 * scale, -4 * scale);
   231→   231→    ctx.lineTo(30 * scale, -2 * scale);
   232→   232→    ctx.lineTo(12 * scale, 6 * scale);
   233→   233→    ctx.closePath();
   234→   234→    ctx.fill();
   235→   235→    // 双角
   236→   236→    ctx.strokeStyle = '#facc15';
   237→   237→    ctx.lineWidth = 2.4;
   238→   238→    ctx.beginPath();
   239→   239→    ctx.moveTo(0, -8 * scale);
   240→   240→    ctx.quadraticCurveTo(-6 * scale, -20 * scale, 4 * scale, -24 * scale);
   241→   241→    ctx.stroke();
   242→   242→    ctx.beginPath();
   243→   243→    ctx.moveTo(-6 * scale, -7 * scale);
   244→   244→    ctx.quadraticCurveTo(-14 * scale, -17 * scale, -6 * scale, -21 * scale);
   245→   245→    ctx.stroke();
   246→   246→    // 龙目
   247→   247→    ctx.fillStyle = '#facc15';
   248→   248→    ctx.shadowColor = '#facc15';
   249→   249→    ctx.shadowBlur = 10;
   250→   250→    ctx.beginPath();
   251→   251→    ctx.arc(10 * scale, -3 * scale, 2.6 * scale, 0, Math.PI * 2);
   252→   252→    ctx.fill();
   253→   253→    ctx.shadowBlur = 0;
   254→   254→    // 龙须
   255→   255→    ctx.strokeStyle = 'rgba(56, 189, 248, 0.85)';
   256→   256→    ctx.lineWidth = 1.6;
   257→   257→    ctx.beginPath();
   258→   258→    ctx.moveTo(-8 * scale, 2 * scale);
   259→   259→    ctx.quadraticCurveTo(-24 * scale, 8 * scale + Math.sin(t * 1.3) * 5, -36 * scale, 2 * scale);
   260→   260→    ctx.stroke();
   261→   261→
   262→   262→    ctx.restore();
   263→   263→    ctx.restore();
   264→   264→
   265→   265→    // 地面掠过阴影
   266→   266→    ctx.save();
   267→   267→    ctx.globalAlpha = 0.3;
   268→   268→    ctx.fillStyle = '#09090b';
   269→   269→    ctx.beginPath();
   270→   270→    ctx.ellipse(screenX, dragon.y + 6, 60 * scale, 14 * scale, 0, 0, Math.PI * 2);
   271→   271→    ctx.fill();
   272→   272→    ctx.restore();
   273→   273→  }

   275→   275→  /** 文房四宝掉落物：毛笔 / 墨锭 / 纸卷 / 砚台，附光柱提示 */
   276→   276→  private drawInkTreasure(t: InkTreasure, cameraX: number, gameTime: number) {
   277→   277→    const screenX = t.x - cameraX;
   278→   278→    if (screenX < -60 || screenX > this.width + 60) return;
   279→   279→
   280→   280→    const ctx = this.ctx;
   281→   281→    const bob = Math.sin(gameTime * 3.2 + t.x * 0.01) * 4;
   282→   282→    const y = t.y - t.altitudeZ - 14 + bob;
   283→   283→
   284→   284→    // 光柱（即将消失时闪烁）
   285→   285→    const fading = t.life < 3 ? 0.4 + Math.sin(gameTime * 10) * 0.3 : 0.85;
   286→   286→    const pillarGrad = ctx.createLinearGradient(screenX, y, screenX, y - 90);
   287→   287→    pillarGrad.addColorStop(0, `rgba(250, 204, 21, ${0.35 * fading})`);
   288→   288→    pillarGrad.addColorStop(1, 'rgba(250, 204, 21, 0)');
   289→   289→    ctx.save();
   290→   290→    ctx.fillStyle = pillarGrad;
   291→   291→    ctx.fillRect(screenX - 13, y - 90, 26, 90);
   292→   292→
   293→   293→    // 底部光环
   294→   294→    ctx.strokeStyle = `rgba(250, 204, 21, ${0.55 * fading})`;
   295→   295→    ctx.lineWidth = 1.6;
   296→   296→    ctx.beginPath();
   297→   297→    ctx.ellipse(screenX, t.y + 2, 20, 7, 0, 0, Math.PI * 2);
   298→   298→    ctx.stroke();
   299→   299→
   300→   300→    ctx.translate(screenX, y);
   301→   301→    ctx.rotate(Math.sin(gameTime * 2) * 0.12);
   302→   302→
   303→   303→    if (t.kind === 'BRUSH') {
   304→   304→      // 毛笔：笔杆 + 笔锋
   305→   305→      ctx.strokeStyle = '#78350f';
   306→   306→      ctx.lineWidth = 3.4;
   307→   307→      ctx.lineCap = 'round';
   308→   308→      ctx.beginPath();
   309→   309→      ctx.moveTo(-7, 9);
   310→   310→      ctx.lineTo(6, -8);
   311→   311→      ctx.stroke();
   312→   312→      ctx.fillStyle = '#09090b';
   313→   313→      ctx.beginPath();
   314→   314→      ctx.moveTo(6, -8);
   315→   315→      ctx.quadraticCurveTo(13, -14, 11, -4);
   316→   316→      ctx.closePath();
   317→   317→      ctx.fill();
   318→   318→      ctx.fillStyle = '#facc15';
   319→   319→      ctx.beginPath();
   320→   320→      ctx.arc(-7, 9, 2.2, 0, Math.PI * 2);
   321→   321→      ctx.fill();
   322→   322→    } else if (t.kind === 'INK') {
   323→   323→      // 墨锭
   324→   324→      ctx.fillStyle = '#18181b';
   325→   325→      ctx.strokeStyle = '#facc15';
   326→   326→      ctx.lineWidth = 1.4;
   327→   327→      ctx.beginPath();
   328→   328→      ctx.roundRect(-8, -6, 16, 12, 2);
   329→   329→      ctx.fill();
   330→   330→      ctx.stroke();
   331→   331→      ctx.fillStyle = '#facc15';
   332→   332→      ctx.font = 'bold 9px "Noto Serif SC", serif';
   333→   333→      ctx.textAlign = 'center';
   334→   334→      ctx.textBaseline = 'middle';
   335→   335→      ctx.fillText('墨', 0, 0.5);
   336→   336→    } else if (t.kind === 'PAPER') {
   337→   337→      // 纸卷
   338→   338→      ctx.save();
   339→   339→      ctx.rotate(-0.5);
   340→   340→      ctx.fillStyle = '#f5f0e6';
   341→   341→      ctx.strokeStyle = '#a8a29e';
   342→   342→      ctx.lineWidth = 1.2;
   343→   343→      ctx.beginPath();
   344→   344→      ctx.roundRect(-10, -5, 20, 10, 2);
   345→   345→      ctx.fill();
   346→   346→      ctx.stroke();
   347→   347→      ctx.strokeStyle = '#78716c';
   348→   348→      ctx.beginPath();
   349→   349→      ctx.moveTo(-6, -1.5);
   350→   350→      ctx.lineTo(6, -1.5);
   351→   351→      ctx.moveTo(-6, 2);
   352→   352→      ctx.lineTo(2, 2);
   353→   353→      ctx.stroke();
   354→   354→      ctx.restore();
   355→   355→    } else {
   356→   356→      // 砚台
   357→   357→      ctx.fillStyle = '#292524';
   358→   358→      ctx.strokeStyle = '#38bdf8';
   359→   359→      ctx.lineWidth = 1.4;
   360→   360→      ctx.beginPath();
   361→   361→      ctx.ellipse(0, 0, 11, 6.5, 0, 0, Math.PI * 2);
   362→   362→      ctx.fill();
   363→   363→      ctx.stroke();
   364→   364→      ctx.fillStyle = '#09090b';
   365→   365→      ctx.beginPath();
   366→   366→      ctx.ellipse(0, -1, 6.5, 3.4, 0, 0, Math.PI * 2);
   367→   367→      ctx.fill();
   368→   368→    }
   369→   369→
   370→   370→    ctx.restore();
   371→   371→  }

   373→   373→  /** 墨滞时刻：屏幕边缘墨紫晕影 + 飘落时之墨点 */
   374→   374→  private drawInkSlowVignette(progress: number, gameTime: number) {
   375→   375→    const ctx = this.ctx;
   376→   376→    const strength = Math.min(1, progress * 1.6);
   377→   377→
   378→   378→    // 四周墨紫晕影
   379→   379→    const vg = ctx.createRadialGradient(
   380→   380→      this.width / 2,
   381→   381→      this.height / 2,
   382→   382→      Math.min(this.width, this.height) * 0.32,
   383→   383→      this.width / 2,
   384→   384→      this.height / 2,
   385→   385→      Math.max(this.width, this.height) * 0.72
   386→   386→    );
   387→   387→    vg.addColorStop(0, 'rgba(76, 29, 149, 0)');
   388→   388→    vg.addColorStop(0.7, `rgba(76, 29, 149, ${0.22 * strength})`);
   389→   389→    vg.addColorStop(1, `rgba(9, 9, 11, ${0.55 * strength})`);
   390→   390→    ctx.save();
   391→   391→    ctx.fillStyle = vg;
   392→   392→    ctx.fillRect(0, 0, this.width, this.height);
   393→   393→
   394→   394→    // 缓落时之墨点
   395→   395→    const dots = 14;
   396→   396→    for (let i = 0; i < dots; i++) {
   397→   397→      const seed = i * 97.13;
   398→   398→      const dx = (Math.sin(seed) * 0.5 + 0.5) * this.width;
   399→   399→      const dy = ((gameTime * 26 + seed * 13) % (this.height + 60)) - 30;
   400→   400→      ctx.globalAlpha = 0.3 * strength;
   401→   401→      ctx.fillStyle = '#c4b5fd';
   402→   402→      ctx.beginPath();
   403→   403→      ctx.arc(dx, dy, 2.2, 0, Math.PI * 2);
   404→   404→      ctx.fill();
   405→   405→    }
   406→   406→    ctx.restore();
   407→   407→  }

   409→   409→  // Draw Sky & Parallax Layer
   410→   410→  private drawSkyAndParallax(cameraX: number, level: LevelConfig, laneMinY: number, gameTime: number) {
   411→   411→    const ctx = this.ctx;
   412→   412→
   413→   413→    // Atmospheric Sky Gradient
   414→   414→    const skyGrad = ctx.createLinearGradient(0, 0, 0, laneMinY);
   415→   415→    if (level.ambientType === 'SNOW') {
   416→   416→      skyGrad.addColorStop(0, '#1c1f24');
   417→   417→      skyGrad.addColorStop(0.6, '#282e38');
   418→   418→      skyGrad.addColorStop(1, '#dfe5ec');
   419→   419→    } else if (level.ambientType === 'BAMBOO') {
   420→   420→      skyGrad.addColorStop(0, '#0c1a16');
   421→   421→      skyGrad.addColorStop(0.5, '#173027');
   422→   422→      skyGrad.addColorStop(1, '#cce3d4');
   423→   423→    } else if (level.ambientType === 'MOON') {
   424→   424→      skyGrad.addColorStop(0, '#070a14');
   425→   425→      skyGrad.addColorStop(0.6, '#13192f');
   426→   426→      skyGrad.addColorStop(1, '#2c3755');
   427→   427→    } else if (level.ambientType === 'PETAL') {
   428→   428→      skyGrad.addColorStop(0, '#1c1218');
   429→   429→      skyGrad.addColorStop(0.6, '#38222b');
   430→   430→      skyGrad.addColorStop(1, '#f2d8dd');
   431→   431→    } else {
   432→   432→      skyGrad.addColorStop(0, '#120d1c');

   449→   449→      let drawW: number, drawH: number;
   450→   450→      if (imgAR > targetAR) {
   451→   451→        drawH = skyH;

   454→   454→        drawW = this.width;

   502→   502→  private drawDistantInkMountains(
   503→   503→    cameraX: number,
   504→   504→    laneMinY: number,
   505→   505→    parallaxSpeed: number,
   506→   506→    color: string,
   507→   507→    alpha: number,
   508→   508→    heightVariance: number
   509→   509→  ) {
   510→   510→    const ctx = this.ctx;
   511→   511→    ctx.save();
   512→   512→    ctx.globalAlpha = alpha;
   513→   513→    ctx.fillStyle = color;
   514→   514→
   515→   515→    const startOffset = -(cameraX * parallaxSpeed);
   516→   516→    ctx.beginPath();
   517→   517→    ctx.moveTo(0, laneMinY);
   518→   518→
   519→   519→    const step = 85;
   520→   520→    const numPoints = Math.ceil(this.width / step) + 2;
   521→   521→    for (let i = 0; i <= numPoints; i++) {
   522→   522→      const x = i * step;
   523→   523→      const worldPos = x - startOffset;
   524→   524→      const y =
   525→   525→        laneMinY - 120 - Math.sin(worldPos * 0.003) * heightVariance - Math.cos(worldPos * 0.007) * 35;
   526→   526→      ctx.lineTo(x, y);
   527→   527→    }
   528→   528→    ctx.lineTo(this.width, laneMinY);
   529→   529→    ctx.closePath();
   530→   530→    ctx.fill();
   531→   531→    ctx.restore();
   532→   532→  }

   582→   582→      ctx.moveTo(0, y);
   583→   583→      ctx.lineTo(this.width, y);

   605→   605→    }
   606→   606→
   607→   607→    ctx.restore();
