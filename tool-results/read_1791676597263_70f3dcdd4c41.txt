     1→import {
     2→  UnitInstance,
     3→  HeroInstance,
     4→  Projectile,
     5→  Particle,
     6→  DamageNumber,
     7→  LevelConfig,
     8→  InkLotusTrap,
     9→  SealStamp,
    10→  InkDragon,
    11→  InkTreasure,
    12→} from '../types/game';
    13→
    14→export class CanvasRenderer {
    15→  private ctx: CanvasRenderingContext2D;
    16→  private width: number;
    17→  private height: number;
    18→  /** 设备像素比：高分屏逻辑坐标 → 物理像素映射，画面更清晰 */
    19→  private dpr: number = 1;
    20→  private bgImage: HTMLImageElement | null = null;
    21→
    22→  constructor(ctx: CanvasRenderingContext2D, width: number, height: number, bgImageSrc?: string) {
    23→    this.ctx = ctx;
    24→    this.width = width;
    25→    this.height = height;
    26→
    27→    if (bgImageSrc) {
    28→      const img = new Image();
    29→      img.src = bgImageSrc;
    30→      img.onload = () => {
    31→        this.bgImage = img;
    32→      };
    33→    }
    34→  }
    35→
    36→  public resize(width: number, height: number, dpr?: number) {
    37→    this.width = width;
    38→    this.height = height;
    39→    if (dpr && dpr > 0) this.dpr = dpr;
    40→  }
    41→
    42→  // 2.5D Coordinate to Screen position with perspective
    43→  private getDepthScale(laneY: number, laneMin: number, laneMax: number): number {
    44→    const t = Math.max(0, Math.min(1, (laneY - laneMin) / (laneMax - laneMin)));
    45→    return 0.86 + t * 0.22; // Depth scale 0.86 (back) to 1.08 (front)
    46→  }
    47→
    48→  // Clear and render entire 2.5D scene with screen shake
    49→  public render(
    50→    cameraX: number,
    51→    level: LevelConfig,
    52→    worldWidth: number,
    53→    laneMinY: number,
    54→    laneMaxY: number,
    55→    playerBaseHp: number,
    56→    playerBaseMaxHp: number,
    57→    enemyBaseHp: number,
    58→    enemyBaseMaxHp: number,
    59→    hero: HeroInstance,
    60→    units: UnitInstance[],
    61→    lotusTraps: InkLotusTrap[],
    62→    projectiles: Projectile[],
    63→    particles: Particle[],
    64→    damageNumbers: DamageNumber[],
    65→    sealStamps: SealStamp[],
    66→    wells: { x: number; y: number; inkRemaining: number }[],
    67→    gameTime: number,
    68→    shakeX: number = 0,
    69→    shakeY: number = 0,
    70→    inkDragons: InkDragon[] = [],
    71→    treasures: InkTreasure[] = [],
    72→    inkSlowProgress: number = 0
    73→  ) {
    74→    const ctx = this.ctx;
    75→    // 每帧重置基准变换到逻辑坐标系（物理分辨率由 dpr 缩放）
    76→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
    77→    ctx.save();
    78→    ctx.translate(shakeX, shakeY);
    79→    ctx.clearRect(0, 0, this.width, this.height);
    80→
    81→    // 1. Sky & Parallax Chinese Mountains
    82→    this.drawSkyAndParallax(cameraX, level, laneMinY, gameTime);
    83→
    84→    // 2. 2.5D Perspective Ground Plane (Courtyard / Riverbank flagstones)
    85→    this.drawPerspectiveGround(cameraX, worldWidth, laneMinY, laneMaxY, level, gameTime);
    86→
    87→    // 3. Ground Lotus Traps / Runes
    88→    lotusTraps.forEach((trap) => {
    89→      this.drawInkLotus(trap, cameraX, laneMinY, laneMaxY, gameTime);
    90→    });
    91→
    92→    // 3.5 文房四宝掉落物（地面发光物）
    93→    treasures.forEach((t) => {
    94→      this.drawInkTreasure(t, cameraX, gameTime);
    95→    });
    96→
    97→    // 4. Ink Wells
    98→    wells.forEach((well) => {
    99→      this.drawInkWell(well.x - cameraX, well.y, laneMinY, laneMaxY, well.inkRemaining, gameTime);
   100→    });
   101→
   102→    // 5. Bases / Citadels
   103→    const baseMidY = (laneMinY + laneMaxY) * 0.5;
   104→    this.drawPlayerCitadel(100 - cameraX, baseMidY, playerBaseHp, playerBaseMaxHp, gameTime);
   105→    this.drawEnemyCitadel(
   106→      worldWidth - 100 - cameraX,
   107→      baseMidY,
   108→      enemyBaseHp,
   109→      enemyBaseMaxHp,
   110→      gameTime,
   111→      level.boss?.name
   112→    );
   113→
   114→    // 6. Draw Hero Inspiring Aura on Ground (Under feet)
   115→    if (hero.state !== 'DEAD') {
   116→      this.drawHeroAura25D(hero, cameraX, laneMinY, laneMaxY, gameTime);
   117→    }
   118→
   119→    // 7. 2.5D Y-Depth Sorting for Characters and Drops
   120→    type RenderItem =
   121→      | { type: 'HERO'; y: number; data: HeroInstance }
   122→      | { type: 'UNIT'; y: number; data: UnitInstance };
   123→
   124→    const renderList: RenderItem[] = [];
   125→    if (hero.state !== 'DEAD') {
   126→      renderList.push({ type: 'HERO', y: hero.y, data: hero });
   127→    }
   128→    units.forEach((u) => {
   129→      renderList.push({ type: 'UNIT', y: u.y, data: u });
   130→    });
   131→
   132→    renderList.sort((a, b) => a.y - b.y);
   133→
   134→    // Render sorted entities with skeletal animation
   135→    renderList.forEach((item) => {
   136→      if (item.type === 'HERO') {
   137→        this.drawSkeletalHero(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   138→      } else {
   139→        this.drawSkeletalUnit(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   140→      }
   141→    });
   142→
   143→    // 7.5 墨龙卷（高连击奖励，飞在最上层）
   144→    inkDragons.forEach((dragon) => {
   145→      this.drawInkDragon(dragon, cameraX, laneMinY, laneMaxY, gameTime);
   146→    });
   147→
   148→    // 8. Projectiles in 2.5D space
   149→    projectiles.forEach((proj) => {
   150→      this.drawProjectile(proj, cameraX, laneMinY, laneMaxY);
   151→    });
   152→
   153→    // 9. Particles
   154→    particles.forEach((p) => {
   155→      this.drawParticle(p, cameraX);
   156→    });
   157→
   158→    // 10. Seal Stamps (Red Calligraphy Stamps)
   159→    sealStamps.forEach((stamp) => {
   160→      this.drawSealStamp(stamp, cameraX);
   161→    });
   162→
   163→    // 11. Floating Damage Numbers
   164→    damageNumbers.forEach((d) => {
   165→      this.drawDamageNumber(d, cameraX);
   166→    });
   167→
   168→    ctx.restore();
   169→
   170→    // 12. 墨滞时刻全屏滤镜（屏幕空间，不受镜头/震动影响）
   171→    if (inkSlowProgress > 0) {
   172→      this.drawInkSlowVignette(inkSlowProgress, gameTime);
   173→    }
   174→  }
   175→
   176→  /** 墨龙卷：蛇形大笔触游走的墨龙 */
   177→  private drawInkDragon(
   178→    dragon: InkDragon,
   179→    cameraX: number,
   180→    laneMinY: number,
   181→    laneMaxY: number,
   182→    gameTime: number
   183→  ) {
   184→    const screenX = dragon.x - cameraX;
   185→    if (screenX < -260 || screenX > this.width + 260) return;
   186→
   187→    const ctx = this.ctx;
   188→    const scale = this.getDepthScale(dragon.y, laneMinY, laneMaxY);
   189→    const t = gameTime * 9;
   190→    const dir = dragon.vx >= 0 ? 1 : -1;
   191→
   192→    ctx.save();
   193→    ctx.translate(screenX, dragon.y - 46 * scale);
   194→
   195→    // 龙身：多段蛇形波浪笔触（越靠尾部越细）
   196→    const segs = 9;
   197→    for (let i = 0; i < segs; i++) {
   198→      const tt = t - i * 0.55;
   199→      const sx = -dir * i * 26 * scale;
   200→      const sy = Math.sin(tt) * 14 * scale;
   201→      const segSize = (20 - i * 1.7) * scale;
   202→      if (segSize <= 1) continue;
   203→
   204→      ctx.save();
   205→      ctx.translate(sx, sy);
   206→      ctx.fillStyle = `rgba(9, 9, 11, ${0.92 - i * 0.07})`;
   207→      ctx.beginPath();
   208→      ctx.ellipse(0, 0, segSize * 1.5, segSize * 0.85, Math.sin(tt) * 0.3, 0, Math.PI * 2);
   209→      ctx.fill();
   210→      // 金墨描边灵光
   211→      ctx.strokeStyle = `rgba(250, 204, 21, ${0.5 - i * 0.05})`;
   212→      ctx.lineWidth = 1.6;
   213→      ctx.stroke();
   214→      ctx.restore();
   215→    }
   216→
   217→    // 龙首：张口吐墨 + 龙角 + 龙须
   218→    const headSway = Math.sin(t) * 6 * scale;
   219→    ctx.save();
   220→    ctx.translate(dir * 10 * scale, headSway);
   221→    ctx.scale(dir, 1);
   222→
   223→    // 头颅
   224→    ctx.fillStyle = '#09090b';
   225→    ctx.beginPath();
   226→    ctx.ellipse(6 * scale, 0, 15 * scale, 10 * scale, 0, 0, Math.PI * 2);
   227→    ctx.fill();
   228→    // 吻部
   229→    ctx.beginPath();
   230→    ctx.moveTo(12 * scale, -4 * scale);
   231→    ctx.lineTo(30 * scale, -2 * scale);
   232→    ctx.lineTo(12 * scale, 6 * scale);
   233→    ctx.closePath();
   234→    ctx.fill();
   235→    // 双角
   236→    ctx.strokeStyle = '#facc15';
   237→    ctx.lineWidth = 2.4;
   238→    ctx.beginPath();
   239→    ctx.moveTo(0, -8 * scale);
   240→    ctx.quadraticCurveTo(-6 * scale, -20 * scale, 4 * scale, -24 * scale);
   241→    ctx.stroke();
   242→    ctx.beginPath();
   243→    ctx.moveTo(-6 * scale, -7 * scale);
   244→    ctx.quadraticCurveTo(-14 * scale, -17 * scale, -6 * scale, -21 * scale);
   245→    ctx.stroke();
   246→    // 龙目
   247→    ctx.fillStyle = '#facc15';
   248→    ctx.shadowColor = '#facc15';
   249→    ctx.shadowBlur = 10;
   250→    ctx.beginPath();
   251→    ctx.arc(10 * scale, -3 * scale, 2.6 * scale, 0, Math.PI * 2);
   252→    ctx.fill();
   253→    ctx.shadowBlur = 0;
   254→    // 龙须
   255→    ctx.strokeStyle = 'rgba(56, 189, 248, 0.85)';
   256→    ctx.lineWidth = 1.6;
   257→    ctx.beginPath();
   258→    ctx.moveTo(-8 * scale, 2 * scale);
   259→    ctx.quadraticCurveTo(-24 * scale, 8 * scale + Math.sin(t * 1.3) * 5, -36 * scale, 2 * scale);
   260→    ctx.stroke();
   261→
   262→    ctx.restore();
   263→    ctx.restore();
   264→
   265→    // 地面掠过阴影
   266→    ctx.save();
   267→    ctx.globalAlpha = 0.3;
   268→    ctx.fillStyle = '#09090b';
   269→    ctx.beginPath();
   270→    ctx.ellipse(screenX, dragon.y + 6, 60 * scale, 14 * scale, 0, 0, Math.PI * 2);
   271→    ctx.fill();
   272→    ctx.restore();
   273→  }

   275→  /** 文房四宝掉落物：毛笔 / 墨锭 / 纸卷 / 砚台，附光柱提示 */
   276→  private drawInkTreasure(t: InkTreasure, cameraX: number, gameTime: number) {
   277→    const screenX = t.x - cameraX;
   278→    if (screenX < -60 || screenX > this.width + 60) return;
   279→
   280→    const ctx = this.ctx;
   281→    const bob = Math.sin(gameTime * 3.2 + t.x * 0.01) * 4;
   282→    const y = t.y - t.altitudeZ - 14 + bob;
   283→
   284→    // 光柱（即将消失时闪烁）
   285→    const fading = t.life < 3 ? 0.4 + Math.sin(gameTime * 10) * 0.3 : 0.85;
   286→    const pillarGrad = ctx.createLinearGradient(screenX, y, screenX, y - 90);
   287→    pillarGrad.addColorStop(0, `rgba(250, 204, 21, ${0.35 * fading})`);
   288→    pillarGrad.addColorStop(1, 'rgba(250, 204, 21, 0)');
   289→    ctx.save();
   290→    ctx.fillStyle = pillarGrad;
   291→    ctx.fillRect(screenX - 13, y - 90, 26, 90);
   292→
   293→    // 底部光环
   294→    ctx.strokeStyle = `rgba(250, 204, 21, ${0.55 * fading})`;
   295→    ctx.lineWidth = 1.6;
   296→    ctx.beginPath();
   297→    ctx.ellipse(screenX, t.y + 2, 20, 7, 0, 0, Math.PI * 2);
   298→    ctx.stroke();
   299→
   300→    ctx.translate(screenX, y);
   301→    ctx.rotate(Math.sin(gameTime * 2) * 0.12);
   302→
   303→    if (t.kind === 'BRUSH') {
   304→      // 毛笔：笔杆 + 笔锋
   305→      ctx.strokeStyle = '#78350f';
   306→      ctx.lineWidth = 3.4;
   307→      ctx.lineCap = 'round';
   308→      ctx.beginPath();
   309→      ctx.moveTo(-7, 9);
   310→      ctx.lineTo(6, -8);
   311→      ctx.stroke();
   312→      ctx.fillStyle = '#09090b';
   313→      ctx.beginPath();
   314→      ctx.moveTo(6, -8);
   315→      ctx.quadraticCurveTo(13, -14, 11, -4);
   316→      ctx.closePath();
   317→      ctx.fill();
   318→      ctx.fillStyle = '#facc15';
   319→      ctx.beginPath();
   320→      ctx.arc(-7, 9, 2.2, 0, Math.PI * 2);
   321→      ctx.fill();
   322→    } else if (t.kind === 'INK') {
   323→      // 墨锭
   324→      ctx.fillStyle = '#18181b';
   325→      ctx.strokeStyle = '#facc15';
   326→      ctx.lineWidth = 1.4;
   327→      ctx.beginPath();
   328→      ctx.roundRect(-8, -6, 16, 12, 2);
   329→      ctx.fill();
   330→      ctx.stroke();
   331→      ctx.fillStyle = '#facc15';
   332→      ctx.font = 'bold 9px "Noto Serif SC", serif';
   333→      ctx.textAlign = 'center';
   334→      ctx.textBaseline = 'middle';
   335→      ctx.fillText('墨', 0, 0.5);
   336→    } else if (t.kind === 'PAPER') {
   337→      // 纸卷
   338→      ctx.save();
   339→      ctx.rotate(-0.5);
   340→      ctx.fillStyle = '#f5f0e6';
   341→      ctx.strokeStyle = '#a8a29e';
   342→      ctx.lineWidth = 1.2;
   343→      ctx.beginPath();
   344→      ctx.roundRect(-10, -5, 20, 10, 2);
   345→      ctx.fill();
   346→      ctx.stroke();
   347→      ctx.strokeStyle = '#78716c';
   348→      ctx.beginPath();
   349→      ctx.moveTo(-6, -1.5);
   350→      ctx.lineTo(6, -1.5);
   351→      ctx.moveTo(-6, 2);
   352→      ctx.lineTo(2, 2);
   353→      ctx.stroke();
   354→      ctx.restore();
   355→    } else {
   356→      // 砚台
   357→      ctx.fillStyle = '#292524';
   358→      ctx.strokeStyle = '#38bdf8';
   359→      ctx.lineWidth = 1.4;
   360→      ctx.beginPath();
   361→      ctx.ellipse(0, 0, 11, 6.5, 0, 0, Math.PI * 2);
   362→      ctx.fill();
   363→      ctx.stroke();
   364→      ctx.fillStyle = '#09090b';
   365→      ctx.beginPath();
   366→      ctx.ellipse(0, -1, 6.5, 3.4, 0, 0, Math.PI * 2);
   367→      ctx.fill();
   368→    }
   369→
   370→    ctx.restore();
   371→  }

   373→  /** 墨滞时刻：屏幕边缘墨紫晕影 + 飘落时之墨点 */
   374→  private drawInkSlowVignette(progress: number, gameTime: number) {
   375→    const ctx = this.ctx;
   376→    const strength = Math.min(1, progress * 1.6);
   377→
   378→    // 四周墨紫晕影
   379→    const vg = ctx.createRadialGradient(
   380→      this.width / 2,
   381→      this.height / 2,
   382→      Math.min(this.width, this.height) * 0.32,
   383→      this.width / 2,
   384→      this.height / 2,
   385→      Math.max(this.width, this.height) * 0.72
   386→    );
   387→    vg.addColorStop(0, 'rgba(76, 29, 149, 0)');
   388→    vg.addColorStop(0.7, `rgba(76, 29, 149, ${0.22 * strength})`);
   389→    vg.addColorStop(1, `rgba(9, 9, 11, ${0.55 * strength})`);
   390→    ctx.save();
   391→    ctx.fillStyle = vg;
   392→    ctx.fillRect(0, 0, this.width, this.height);
   393→
   394→    // 缓落时之墨点
   395→    const dots = 14;
   396→    for (let i = 0; i < dots; i++) {
   397→      const seed = i * 97.13;
   398→      const dx = (Math.sin(seed) * 0.5 + 0.5) * this.width;
   399→      const dy = ((gameTime * 26 + seed * 13) % (this.height + 60)) - 30;
   400→      ctx.globalAlpha = 0.3 * strength;
   401→      ctx.fillStyle = '#c4b5fd';
   402→      ctx.beginPath();
   403→      ctx.arc(dx, dy, 2.2, 0, Math.PI * 2);
   404→      ctx.fill();
   405→    }
   406→    ctx.restore();
   407→  }

   409→  // Draw Sky & Parallax Layer
   410→  private drawSkyAndParallax(cameraX: number, level: LevelConfig, laneMinY: number, gameTime: number) {
   411→    const ctx = this.ctx;
   412→
   413→    // Atmospheric Sky Gradient
   414→    const skyGrad = ctx.createLinearGradient(0, 0, 0, laneMinY);
   415→    if (level.ambientType === 'SNOW') {
   416→      skyGrad.addColorStop(0, '#1c1f24');
   417→      skyGrad.addColorStop(0.6, '#282e38');
   418→      skyGrad.addColorStop(1, '#dfe5ec');
   419→    } else if (level.ambientType === 'BAMBOO') {
   420→      skyGrad.addColorStop(0, '#0c1a16');
   421→      skyGrad.addColorStop(0.5, '#173027');
   422→      skyGrad.addColorStop(1, '#cce3d4');
   423→    } else if (level.ambientType === 'MOON') {
   424→      skyGrad.addColorStop(0, '#070a14');
   425→      skyGrad.addColorStop(0.6, '#13192f');
   426→      skyGrad.addColorStop(1, '#2c3755');
   427→    } else if (level.ambientType === 'PETAL') {
   428→      skyGrad.addColorStop(0, '#1c1218');
   429→      skyGrad.addColorStop(0.6, '#38222b');
   430→      skyGrad.addColorStop(1, '#f2d8dd');
   431→    } else {
   432→      skyGrad.addColorStop(0, '#120d1c');
   433→      skyGrad.addColorStop(0.4, '#2e192f');
   434→      skyGrad.addColorStop(0.8, '#6b3d43');
   435→      skyGrad.addColorStop(1, '#ebd0ab');
   436→    }
   437→    ctx.fillStyle = skyGrad;
   438→    ctx.fillRect(0, 0, this.width, laneMinY + 40);
   439→
   440→    // Background Image Art（等比 cover 绘制：窄屏/宽屏均不拉伸变形，底部对齐地平线，横向平铺视差）
   441→    if (this.bgImage && this.bgImage.naturalWidth > 0) {
   442→      ctx.save();
   443→      ctx.globalAlpha = 0.25;
   444→      const img = this.bgImage;
   445→      const skyH = laneMinY + 60;
   446→      const imgAR = img.naturalWidth / img.naturalHeight;
   447→      const targetAR = this.width / Math.max(1, skyH);
   448→      // cover 逻辑：铺满目标区域所需的最小等比缩放
   449→      let drawW: number, drawH: number;
   450→      if (imgAR > targetAR) {
   451→        drawH = skyH;
   452→        drawW = skyH * imgAR;
   453→      } else {
   454→        drawW = this.width;
   455→        drawH = this.width / imgAR;
   456→      }
   457→      // 底部对齐地平线（山景根部贴地），超高部分向天空裁切
   458→      const drawY = skyH - drawH;
   459→      // 横向平铺 + 视差滚动
   460→      const tileW = drawW;
   461→      const bgOffset = -((cameraX * 0.15) % tileW);
   462→      for (let x = bgOffset; x < this.width; x += tileW) {
   463→        ctx.drawImage(img, x, drawY, tileW, drawH);
   464→      }
   465→      ctx.restore();
   466→    }
   467→
   468→    // Celestial Body
   469→    const celestialX = this.width * 0.75 - cameraX * 0.05;
   470→    ctx.save();
   471→    if (level.ambientType === 'MOON') {
   472→      const moonGlow = ctx.createRadialGradient(celestialX, 85, 10, celestialX, 85, 75);
   473→      moonGlow.addColorStop(0, 'rgba(255, 255, 240, 0.95)');
   474→      moonGlow.addColorStop(0.4, 'rgba(230, 240, 255, 0.4)');
   475→      moonGlow.addColorStop(1, 'rgba(200, 220, 255, 0)');
   476→      ctx.fillStyle = moonGlow;
   477→      ctx.beginPath();
   478→      ctx.arc(celestialX, 85, 75, 0, Math.PI * 2);
   479→      ctx.fill();
   480→
   481→      ctx.fillStyle = '#fffef0';
   482→      ctx.beginPath();
   483→      ctx.arc(celestialX, 85, 30, 0, Math.PI * 2);
   484→      ctx.fill();
   485→    } else if (level.ambientType === 'DAWN') {
   486→      const sunGlow = ctx.createRadialGradient(celestialX, 100, 15, celestialX, 100, 130);
   487→      sunGlow.addColorStop(0, 'rgba(255, 220, 140, 0.9)');
   488→      sunGlow.addColorStop(0.3, 'rgba(240, 160, 90, 0.4)');
   489→      sunGlow.addColorStop(1, 'rgba(230, 100, 70, 0)');
   490→      ctx.fillStyle = sunGlow;
   491→      ctx.beginPath();
   492→      ctx.arc(celestialX, 100, 130, 0, Math.PI * 2);
   493→      ctx.fill();
   494→    }
   495→    ctx.restore();
   496→
   497→    // Distant Ink Mountains
   498→    this.drawDistantInkMountains(cameraX, laneMinY, 0.2, '#333b47', 0.35, 120);
   499→    this.drawDistantInkMountains(cameraX, laneMinY, 0.45, '#1e242c', 0.6, 75);
   500→  }

   502→  private drawDistantInkMountains(
   503→    cameraX: number,
   504→    laneMinY: number,
   505→    parallaxSpeed: number,
   506→    color: string,
   507→    alpha: number,
   508→    heightVariance: number
   509→  ) {
   510→    const ctx = this.ctx;
   511→    ctx.save();
   512→    ctx.globalAlpha = alpha;
   513→    ctx.fillStyle = color;
   514→
   515→    const startOffset = -(cameraX * parallaxSpeed);
   516→    ctx.beginPath();
   517→    ctx.moveTo(0, laneMinY);
   518→
   519→    const step = 85;
   520→    const numPoints = Math.ceil(this.width / step) + 2;
   521→    for (let i = 0; i <= numPoints; i++) {
   522→      const x = i * step;
   523→      const worldPos = x - startOffset;
   524→      const y =
   525→        laneMinY - 120 - Math.sin(worldPos * 0.003) * heightVariance - Math.cos(worldPos * 0.007) * 35;
   526→      ctx.lineTo(x, y);
   527→    }
   528→    ctx.lineTo(this.width, laneMinY);
   529→    ctx.closePath();
   530→    ctx.fill();
   531→    ctx.restore();
   532→  }

   534→  // Draw 2.5D Perspective Ground Plane
   535→  private drawPerspectiveGround(
   536→    cameraX: number,
   537→    worldWidth: number,
   538→    laneMinY: number,
   539→    laneMaxY: number,
   540→    level: LevelConfig,
   541→    gameTime: number
   542→  ) {
   543→    const ctx = this.ctx;
   544→
   545→    // Perspective depth gradient across the walkable lane
   546→    const groundGrad = ctx.createLinearGradient(0, laneMinY, 0, this.height);
   547→    if (level.ambientType === 'SNOW') {
   548→      groundGrad.addColorStop(0, '#cbd5e1');
   549→      groundGrad.addColorStop(0.3, '#94a3b8');
   550→      groundGrad.addColorStop(1, '#334155');
   551→    } else if (level.ambientType === 'BAMBOO') {
   552→      groundGrad.addColorStop(0, '#334835');
   553→      groundGrad.addColorStop(0.3, '#213023');
   554→      groundGrad.addColorStop(1, '#111912');
   555→    } else {
   556→      groundGrad.addColorStop(0, '#26262b');
   557→      groundGrad.addColorStop(0.3, '#1c1c20');
   558→      groundGrad.addColorStop(1, '#0e0e11');
   559→    }
   560→
   561→    ctx.fillStyle = groundGrad;
   562→    ctx.fillRect(0, laneMinY, this.width, this.height - laneMinY);
   563→
   564→    // Traditional Calligraphy Ink Horizon Line
   565→    ctx.save();
   566→    ctx.strokeStyle = '#09090b';
   567→    ctx.lineWidth = 3.5;
   568→    ctx.beginPath();
   569→    ctx.moveTo(0, laneMinY);
   570→    ctx.lineTo(this.width, laneMinY);
   571→    ctx.stroke();
   572→
   573→    // 2.5D Flagstone grid lines (isometric depth perspective)
   574→    ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
   575→    ctx.lineWidth = 1;
   576→
   577→    // Horizontal depth division lines
   578→    const depthSteps = 4;
   579→    for (let i = 1; i <= depthSteps; i++) {
   580→      const y = laneMinY + ((laneMaxY - laneMinY) / depthSteps) * i;
   581→      ctx.beginPath();
   582→      ctx.moveTo(0, y);
   583→      ctx.lineTo(this.width, y);
   584→      ctx.stroke();
   585→    }
   586→
   587→    // Vertical diagonal grid lines
   588→    const tileSpacing = 160;
   589→    const tileOffset = -(cameraX) % tileSpacing;
   590→    for (let x = tileOffset; x < this.width + tileSpacing; x += tileSpacing) {
   591→      ctx.beginPath();
   592→      ctx.moveTo(x - 30, laneMinY);
   593→      ctx.lineTo(x + 50, laneMaxY);
   594→      ctx.stroke();
   595→    }
   596→
   597→    // Water puddle reflections along the riverbank
   598→    const puddleOffset = -(cameraX * 0.9) % 320;
   599→    for (let px = puddleOffset; px < this.width; px += 320) {
   600→      const puddleY = laneMinY + 65;
   601→      ctx.fillStyle = 'rgba(56, 189, 248, 0.06)';
   602→      ctx.beginPath();
   603→      ctx.ellipse(px, puddleY, 35, 10, 0, 0, Math.PI * 2);
   604→      ctx.fill();
   605→    }
   606→
   607→    ctx.restore();
   608→  }

   709→  // Enhanced Multi-Joint Skeletal Hero Animation
   710→  private drawSkeletalHero(
   711→    hero: HeroInstance,
   712→    cameraX: number,
   713→    laneMinY: number,
   714→    laneMaxY: number,
   715→    gameTime: number
   716→  ) {
   717→    const screenX = hero.x - cameraX;
   718→    if (screenX < -120 || screenX > this.width + 120) return;
   719→
   720→    const ctx = this.ctx;
   721→    const groundY = hero.y;
   722→    const altitudeZ = hero.altitudeZ || 0;
   723→    const heroY = groundY - altitudeZ;
   724→    const scale = this.getDepthScale(hero.y, laneMinY, laneMaxY);
   725→    const facing = hero.facing;
   726→
   727→    // 1. Draw 2.5D Drop Shadow on ground
   728→    this.drawDropShadow(screenX, groundY, 24 * scale, altitudeZ);
   729→
   730→    ctx.save();
   731→    ctx.translate(screenX, heroY);
   732→    ctx.scale(facing * scale, scale);
   733→
   734→    // Flinch Recoil if taking damage
   735→    if (hero.flinchTimer > 0) {
   736→      ctx.rotate(-0.15);
   737→      ctx.translate(-5, 0);
   738→    }
   739→
   740→    // Skeletal Gait Calculation
   741→    const isMoving = hero.state === 'WALK';
   742→    const walkFreq = 12;
   743→    const gaitPhase = isMoving ? gameTime * walkFreq : 0;
   744→
   745→    // Pelvis & Body Bobbing
   746→    const hipBob = isMoving ? Math.abs(Math.sin(gaitPhase)) * 4 : Math.sin(gameTime * 2) * 1.5;
   747→    const hipY = -28 + hipBob;
   748→
   749→    // 2. Billowing Multi-Segment Silk Cape (Fluid physics)
   750→    this.drawMultiSegmentCape(ctx, hipY - 20, isMoving, facing, hero.state, gameTime);
   751→
   752→    // 3. Multi-Joint Legs (Thigh -> Knee -> Shin -> Foot)
   753→    ctx.strokeStyle = '#09090b';
   754→    ctx.lineWidth = 4;
   755→    ctx.lineCap = 'round';
   756→    ctx.lineJoin = 'round';
   757→
   758→    const leg1Swing = isMoving ? Math.sin(gaitPhase) * 0.7 : 0.05;
   759→    const leg2Swing = isMoving ? -Math.sin(gaitPhase) * 0.7 : -0.05;
   760→
   761→    // Left Leg (Back)
   762→    this.drawLimbBone(ctx, -2, hipY, leg1Swing, 14, 14);
   763→
   764→    // Right Leg (Front)
   765→    this.drawLimbBone(ctx, 2, hipY, leg2Swing, 14, 14);
   766→
   767→    // 4. Torso & Robes (Spine bone)
   768→    const spineLean = isMoving ? 0.12 : 0;
   769→    const shoulderX = Math.sin(spineLean) * 22;
   770→    const shoulderY = hipY - 22;
   771→
   772→    ctx.beginPath();
   773→    ctx.moveTo(0, hipY);
   774→    ctx.lineTo(shoulderX, shoulderY);
   775→    ctx.stroke();
   776→
   777→    // Robe Sash / Belt in jade turquoise
   778→    ctx.fillStyle = '#06b6d4';
   779→    ctx.fillRect(-6, hipY - 4, 12, 4);
   780→
   781→    // 5. Head, Neck & Topknot
   782→    const headX = shoulderX + 2;
   783→    const headY = shoulderY - 14;
   784→
   785→    ctx.fillStyle = '#09090b';
   786→    ctx.beginPath();
   787→    ctx.arc(headX, headY, 8.5, 0, Math.PI * 2);
   788→    ctx.fill();
   789→
   790→    // Topknot & Flowing Ribbon
   791→    ctx.beginPath();
   792→    ctx.arc(headX, headY - 10, 4, 0, Math.PI * 2);
   793→    ctx.fill();
   794→
   795→    ctx.strokeStyle = '#06b6d4';
   796→    ctx.lineWidth = 2.2;
   797→    const ribbonWave = Math.sin(gameTime * 6) * 6;
   798→    ctx.beginPath();
   799→    ctx.moveTo(headX, headY - 10);
   800→    ctx.quadraticCurveTo(headX - 12 + ribbonWave * 0.5, headY - 12, headX - 22 + ribbonWave, headY - 6);
   801→    ctx.stroke();
   802→
   803→    // 6. Multi-Joint Arms & Luminous Brush-Sword
   804→    ctx.strokeStyle = '#09090b';
   805→    ctx.lineWidth = 4;
   806→
   807→    // Handle Attack animations
   808→    if (hero.state === 'ATTACK_1') {
   809→      // Horizontal sweeping slash
   810→      this.drawLimbBone(ctx, shoulderX, shoulderY, -0.4, 16, 16);
   811→      this.drawBrushSword(ctx, shoulderX + 28, shoulderY - 8, 0.2, true);
   812→      this.drawInkSlashArc(ctx, 40, shoulderY - 8, 65, -0.6, 0.6);
   813→    } else if (hero.state === 'ATTACK_2') {
   814→      // Upward calligraphy stroke
   815→      this.drawLimbBone(ctx, shoulderX, shoulderY, -1.2, 16, 16);
   816→      this.drawBrushSword(ctx, shoulderX + 22, shoulderY - 26, -1.1, true);
   817→      this.drawInkSlashArc(ctx, 35, shoulderY - 26, 70, -1.3, 0.1);
   818→    } else if (hero.state === 'ATTACK_3') {
   819→      // Plunging heavy stroke
   820→      this.drawLimbBone(ctx, shoulderX, shoulderY, 0.6, 16, 16);
   821→      this.drawBrushSword(ctx, shoulderX + 32, shoulderY + 8, 0.9, true);
   822→      this.drawInkSlashArc(ctx, 40, shoulderY + 8, 80, 0.2, 1.4);
   823→    } else if (hero.state === 'ATTACK_AERIAL') {
   824→      // Airborne diving plunge
   825→      this.drawLimbBone(ctx, shoulderX, shoulderY, 1.4, 16, 16);
   826→      this.drawBrushSword(ctx, shoulderX + 20, shoulderY + 22, 1.57, true);
   827→    } else if (hero.state === 'SKILL_DASH') {
   828→      // Forward thrust
   829→      this.drawLimbBone(ctx, shoulderX, shoulderY, -0.1, 16, 16);
   830→      this.drawBrushSword(ctx, shoulderX + 36, shoulderY - 4, 0, true);
   831→    } else {
   832→      // Idle / Walking holding sword gracefully
   833→      const armSway = isMoving ? Math.sin(gaitPhase) * 0.3 : 0;
   834→      this.drawLimbBone(ctx, shoulderX, shoulderY, 0.5 + armSway, 14, 14);
   835→      this.drawBrushSword(ctx, shoulderX + 16, shoulderY + 12, 0.7, false);
   836→    }
   837→
   838→    // Shield Dome if active
   839→    if (hero.shieldHp > 0) {
   840→      ctx.save();
   841→      ctx.strokeStyle = 'rgba(56, 189, 248, 0.85)';
   842→      ctx.lineWidth = 2.5;
   843→      ctx.setLineDash([5, 5]);
   844→      ctx.beginPath();
   845→      ctx.arc(0, hipY - 10, 42, 0, Math.PI * 2);
   846→      ctx.stroke();
   847→      ctx.restore();
   848→    }
   849→
   850→    ctx.restore();
   851→
   852→    // Overhead HUD
   853→    this.drawHeroOverheadHUD(screenX, heroY - 82 * scale, hero, scale);
   854→  }
