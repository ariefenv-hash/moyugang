     1→import {
     2→  UnitInstance,
     3→  HeroInstance,
     4→  Projectile,
     5→  Particle,
     6→  DamageNumber,
     7→  LevelConfig,
     8→  InkLotusTrap,
     9→  SealStamp,
    10→} from '../types/game';
    11→
    12→export class CanvasRenderer {
    13→  private ctx: CanvasRenderingContext2D;
    14→  private width: number;
    15→  private height: number;
    16→  /** 设备像素比：高分屏逻辑坐标 → 物理像素映射，画面更清晰 */
    17→  private dpr: number = 1;
    18→  private bgImage: HTMLImageElement | null = null;
    19→
    20→  constructor(ctx: CanvasRenderingContext2D, width: number, height: number, bgImageSrc?: string) {
    21→    this.ctx = ctx;
    22→    this.width = width;
    23→    this.height = height;
    24→
    25→    if (bgImageSrc) {
    26→      const img = new Image();
    27→      img.src = bgImageSrc;
    28→      img.onload = () => {
    29→        this.bgImage = img;
    30→      };
    31→    }
    32→  }
    33→
    34→  public resize(width: number, height: number, dpr?: number) {
    35→    this.width = width;
    36→    this.height = height;
    37→    if (dpr && dpr > 0) this.dpr = dpr;
    38→  }
    39→
    40→  // 2.5D Coordinate to Screen position with perspective
    41→  private getDepthScale(laneY: number, laneMin: number, laneMax: number): number {
    42→    const t = Math.max(0, Math.min(1, (laneY - laneMin) / (laneMax - laneMin)));
    43→    return 0.86 + t * 0.22; // Depth scale 0.86 (back) to 1.08 (front)
    44→  }
    45→
    46→  // Clear and render entire 2.5D scene with screen shake
    47→  public render(
    48→    cameraX: number,
    49→    level: LevelConfig,
    50→    worldWidth: number,
    51→    laneMinY: number,
    52→    laneMaxY: number,
    53→    playerBaseHp: number,
    54→    playerBaseMaxHp: number,
    55→    enemyBaseHp: number,
    56→    enemyBaseMaxHp: number,
    57→    hero: HeroInstance,
    58→    units: UnitInstance[],
    59→    lotusTraps: InkLotusTrap[],
    60→    projectiles: Projectile[],
    61→    particles: Particle[],
    62→    damageNumbers: DamageNumber[],
    63→    sealStamps: SealStamp[],
    64→    wells: { x: number; y: number; inkRemaining: number }[],
    65→    gameTime: number,
    66→    shakeX: number = 0,
    67→    shakeY: number = 0
    68→  ) {
    69→    const ctx = this.ctx;
    70→    // 每帧重置基准变换到逻辑坐标系（物理分辨率由 dpr 缩放）
    71→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
    72→    ctx.save();
    73→    ctx.translate(shakeX, shakeY);
    74→    ctx.clearRect(0, 0, this.width, this.height);
    75→
    76→    // 1. Sky & Parallax Chinese Mountains
    77→    this.drawSkyAndParallax(cameraX, level, laneMinY, gameTime);
    78→
    79→    // 2. 2.5D Perspective Ground Plane (Courtyard / Riverbank flagstones)
    80→    this.drawPerspectiveGround(cameraX, worldWidth, laneMinY, laneMaxY, level, gameTime);
    81→
    82→    // 3. Ground Lotus Traps / Runes
    83→    lotusTraps.forEach((trap) => {
    84→      this.drawInkLotus(trap, cameraX, laneMinY, laneMaxY, gameTime);
    85→    });
    86→
    87→    // 4. Ink Wells
    88→    wells.forEach((well) => {
    89→      this.drawInkWell(well.x - cameraX, well.y, laneMinY, laneMaxY, well.inkRemaining, gameTime);
    90→    });
    91→
    92→    // 5. Bases / Citadels
    93→    const baseMidY = (laneMinY + laneMaxY) * 0.5;
    94→    this.drawPlayerCitadel(100 - cameraX, baseMidY, playerBaseHp, playerBaseMaxHp, gameTime);
    95→    this.drawEnemyCitadel(
    96→      worldWidth - 100 - cameraX,
    97→      baseMidY,
    98→      enemyBaseHp,
    99→      enemyBaseMaxHp,
   100→      gameTime,
   101→      level.boss?.name
   102→    );
   103→
   104→    // 6. Draw Hero Inspiring Aura on Ground (Under feet)
   105→    if (hero.state !== 'DEAD') {
   106→      this.drawHeroAura25D(hero, cameraX, laneMinY, laneMaxY, gameTime);
   107→    }
   108→
   109→    // 7. 2.5D Y-Depth Sorting for Characters and Drops
   110→    type RenderItem =
   111→      | { type: 'HERO'; y: number; data: HeroInstance }
   112→      | { type: 'UNIT'; y: number; data: UnitInstance };
   113→
   114→    const renderList: RenderItem[] = [];
   115→    if (hero.state !== 'DEAD') {
   116→      renderList.push({ type: 'HERO', y: hero.y, data: hero });
   117→    }
   118→    units.forEach((u) => {
   119→      renderList.push({ type: 'UNIT', y: u.y, data: u });
   120→    });
   121→
   122→    renderList.sort((a, b) => a.y - b.y);
   123→
   124→    // Render sorted entities with skeletal animation
   125→    renderList.forEach((item) => {
   126→      if (item.type === 'HERO') {
   127→        this.drawSkeletalHero(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   128→      } else {
   129→        this.drawSkeletalUnit(item.data, cameraX, laneMinY, laneMaxY, gameTime);
   130→      }
   131→    });
   132→
   133→    // 8. Projectiles in 2.5D space
   134→    projectiles.forEach((proj) => {
   135→      this.drawProjectile(proj, cameraX, laneMinY, laneMaxY);
   136→    });
   137→
   138→    // 9. Particles
   139→    particles.forEach((p) => {
   140→      this.drawParticle(p, cameraX);
   141→    });
   142→
   143→    // 10. Seal Stamps (Red Calligraphy Stamps)
   144→    sealStamps.forEach((stamp) => {
   145→      this.drawSealStamp(stamp, cameraX);
   146→    });
   147→
   148→    // 11. Floating Damage Numbers
   149→    damageNumbers.forEach((d) => {
   150→      this.drawDamageNumber(d, cameraX);
   151→    });
   152→
   153→    ctx.restore();
   154→  }

   156→  // Draw Sky & Parallax Layer
   157→  private drawSkyAndParallax(cameraX: number, level: LevelConfig, laneMinY: number, gameTime: number) {
   158→    const ctx = this.ctx;
   159→
   160→    // Atmospheric Sky Gradient
   161→    const skyGrad = ctx.createLinearGradient(0, 0, 0, laneMinY);
   162→    if (level.ambientType === 'SNOW') {
   163→      skyGrad.addColorStop(0, '#1c1f24');
   164→      skyGrad.addColorStop(0.6, '#282e38');
   165→      skyGrad.addColorStop(1, '#dfe5ec');
   166→    } else if (level.ambientType === 'BAMBOO') {
   167→      skyGrad.addColorStop(0, '#0c1a16');
   168→      skyGrad.addColorStop(0.5, '#173027');
   169→      skyGrad.addColorStop(1, '#cce3d4');
   170→    } else if (level.ambientType === 'MOON') {
   171→      skyGrad.addColorStop(0, '#070a14');
   172→      skyGrad.addColorStop(0.6, '#13192f');
   173→      skyGrad.addColorStop(1, '#2c3755');
   174→    } else if (level.ambientType === 'PETAL') {
   175→      skyGrad.addColorStop(0, '#1c1218');
   176→      skyGrad.addColorStop(0.6, '#38222b');
   177→      skyGrad.addColorStop(1, '#f2d8dd');
   178→    } else {
   179→      skyGrad.addColorStop(0, '#120d1c');
   180→      skyGrad.addColorStop(0.4, '#2e192f');
   181→      skyGrad.addColorStop(0.8, '#6b3d43');
   182→      skyGrad.addColorStop(1, '#ebd0ab');
   183→    }
   184→    ctx.fillStyle = skyGrad;
   185→    ctx.fillRect(0, 0, this.width, laneMinY + 40);
   186→
   187→    // Background Image Art
   188→    if (this.bgImage) {
   189→      ctx.save();
   190→      ctx.globalAlpha = 0.25;
   191→      const bgOffset = -(cameraX * 0.15) % this.width;
   192→      ctx.drawImage(this.bgImage, bgOffset, 0, this.width, laneMinY + 60);
   193→      ctx.drawImage(this.bgImage, bgOffset + this.width, 0, this.width, laneMinY + 60);
   194→      ctx.restore();
   195→    }
   196→
   197→    // Celestial Body
   198→    const celestialX = this.width * 0.75 - cameraX * 0.05;
   199→    ctx.save();
   200→    if (level.ambientType === 'MOON') {
   201→      const moonGlow = ctx.createRadialGradient(celestialX, 85, 10, celestialX, 85, 75);
   202→      moonGlow.addColorStop(0, 'rgba(255, 255, 240, 0.95)');
   203→      moonGlow.addColorStop(0.4, 'rgba(230, 240, 255, 0.4)');
   204→      moonGlow.addColorStop(1, 'rgba(200, 220, 255, 0)');
   205→      ctx.fillStyle = moonGlow;
   206→      ctx.beginPath();
   207→      ctx.arc(celestialX, 85, 75, 0, Math.PI * 2);
   208→      ctx.fill();
   209→
   210→      ctx.fillStyle = '#fffef0';
   211→      ctx.beginPath();
   212→      ctx.arc(celestialX, 85, 30, 0, Math.PI * 2);
   213→      ctx.fill();
   214→    } else if (level.ambientType === 'DAWN') {
   215→      const sunGlow = ctx.createRadialGradient(celestialX, 100, 15, celestialX, 100, 130);
   216→      sunGlow.addColorStop(0, 'rgba(255, 220, 140, 0.9)');
   217→      sunGlow.addColorStop(0.3, 'rgba(240, 160, 90, 0.4)');
   218→      sunGlow.addColorStop(1, 'rgba(230, 100, 70, 0)');
   219→      ctx.fillStyle = sunGlow;
   220→      ctx.beginPath();
   221→      ctx.arc(celestialX, 100, 130, 0, Math.PI * 2);
   222→      ctx.fill();
   223→    }
   224→    ctx.restore();
   225→
   226→    // Distant Ink Mountains
   227→    this.drawDistantInkMountains(cameraX, laneMinY, 0.2, '#333b47', 0.35, 120);
   228→    this.drawDistantInkMountains(cameraX, laneMinY, 0.45, '#1e242c', 0.6, 75);
   229→  }
   230→
   231→  private drawDistantInkMountains(
   232→    cameraX: number,
   233→    laneMinY: number,
   234→    parallaxSpeed: number,
   235→    color: string,
   236→    alpha: number,
   237→    heightVariance: number
   238→  ) {
   239→    const ctx = this.ctx;
   240→    ctx.save();
   241→    ctx.globalAlpha = alpha;
   242→    ctx.fillStyle = color;
   243→
   244→    const startOffset = -(cameraX * parallaxSpeed);
   245→    ctx.beginPath();
   246→    ctx.moveTo(0, laneMinY);
   247→
   248→    const step = 85;
   249→    const numPoints = Math.ceil(this.width / step) + 2;
   250→    for (let i = 0; i <= numPoints; i++) {
   251→      const x = i * step;
   252→      const worldPos = x - startOffset;
   253→      const y =
   254→        laneMinY - 120 - Math.sin(worldPos * 0.003) * heightVariance - Math.cos(worldPos * 0.007) * 35;
   255→      ctx.lineTo(x, y);
   256→    }
   257→    ctx.lineTo(this.width, laneMinY);
   258→    ctx.closePath();
   259→    ctx.fill();
   260→    ctx.restore();
   261→  }

   263→  // Draw 2.5D Perspective Ground Plane
   264→  private drawPerspectiveGround(
   265→    cameraX: number,
   266→    worldWidth: number,
   267→    laneMinY: number,
   268→    laneMaxY: number,
   269→    level: LevelConfig,
   270→    gameTime: number
   271→  ) {
   272→    const ctx = this.ctx;
   273→
   274→    // Perspective depth gradient across the walkable lane
   275→    const groundGrad = ctx.createLinearGradient(0, laneMinY, 0, this.height);
   276→    if (level.ambientType === 'SNOW') {
   277→      groundGrad.addColorStop(0, '#cbd5e1');
   278→      groundGrad.addColorStop(0.3, '#94a3b8');
   279→      groundGrad.addColorStop(1, '#334155');
   280→    } else if (level.ambientType === 'BAMBOO') {
   281→      groundGrad.addColorStop(0, '#334835');
   282→      groundGrad.addColorStop(0.3, '#213023');
   283→      groundGrad.addColorStop(1, '#111912');
   284→    } else {
   285→      groundGrad.addColorStop(0, '#26262b');
   286→      groundGrad.addColorStop(0.3, '#1c1c20');
   287→      groundGrad.addColorStop(1, '#0e0e11');
   288→    }
   289→
   290→    ctx.fillStyle = groundGrad;
   291→    ctx.fillRect(0, laneMinY, this.width, this.height - laneMinY);
   292→
   293→    // Traditional Calligraphy Ink Horizon Line
   294→    ctx.save();
   295→    ctx.strokeStyle = '#09090b';
   296→    ctx.lineWidth = 3.5;
   297→    ctx.beginPath();
   298→    ctx.moveTo(0, laneMinY);
   299→    ctx.lineTo(this.width, laneMinY);
   300→    ctx.stroke();
   301→
   302→    // 2.5D Flagstone grid lines (isometric depth perspective)
   303→    ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
   304→    ctx.lineWidth = 1;
   305→
   306→    // Horizontal depth division lines
   307→    const depthSteps = 4;
   308→    for (let i = 1; i <= depthSteps; i++) {
   309→      const y = laneMinY + ((laneMaxY - laneMinY) / depthSteps) * i;
   310→      ctx.beginPath();
   311→      ctx.moveTo(0, y);
   312→      ctx.lineTo(this.width, y);
   313→      ctx.stroke();
   314→    }
   315→
   316→    // Vertical diagonal grid lines
   317→    const tileSpacing = 160;
   318→    const tileOffset = -(cameraX) % tileSpacing;
   319→    for (let x = tileOffset; x < this.width + tileSpacing; x += tileSpacing) {
   320→      ctx.beginPath();
   321→      ctx.moveTo(x - 30, laneMinY);
   322→      ctx.lineTo(x + 50, laneMaxY);
   323→      ctx.stroke();
   324→    }
   325→
   326→    // Water puddle reflections along the riverbank
   327→    const puddleOffset = -(cameraX * 0.9) % 320;
   328→    for (let px = puddleOffset; px < this.width; px += 320) {
   329→      const puddleY = laneMinY + 65;
   330→      ctx.fillStyle = 'rgba(56, 189, 248, 0.06)';
   331→      ctx.beginPath();
   332→      ctx.ellipse(px, puddleY, 35, 10, 0, 0, Math.PI * 2);
   333→      ctx.fill();
   334→    }
   335→
   336→    ctx.restore();
   337→  }

   339→  // Draw 2.5D Drop Shadow under characters
   340→  private drawDropShadow(screenX: number, groundY: number, radiusX: number, altitudeZ: number) {
   341→    const ctx = this.ctx;
   342→    ctx.save();
   343→
   344→    // Shadow gets fainter and smaller when jumping high
   345→    const shadowFactor = Math.max(0.2, 1 - altitudeZ / 180);
   346→    const radX = radiusX * shadowFactor;
   347→    const radY = radX * 0.35;
   348→
   349→    ctx.fillStyle = `rgba(0, 0, 0, ${0.45 * shadowFactor})`;
   350→    ctx.beginPath();
   351→    ctx.ellipse(screenX, groundY, radX, radY, 0, 0, Math.PI * 2);
   352→    ctx.fill();
   353→    ctx.restore();
   354→  }

   356→  // Draw Hero Inspiring Aura (风雅领域) in 2.5D
   357→  private drawHeroAura25D(
   358→    hero: HeroInstance,
   359→    cameraX: number,
   360→    laneMinY: number,
   361→    laneMaxY: number,
   362→    gameTime: number
   363→  ) {
   364→    const screenX = hero.x - cameraX;
   365→    const groundY = hero.y;
   366→    const ctx = this.ctx;
   367→    ctx.save();
   368→
   369→    const scale = this.getDepthScale(hero.y, laneMinY, laneMaxY);
   370→    const radiusX = 220 * scale;
   371→    const radiusY = radiusX * 0.35;
   372→
   373→    // Glowing ground ellipse
   374→    ctx.beginPath();
   375→    ctx.ellipse(screenX, groundY, radiusX, radiusY, 0, 0, Math.PI * 2);
   376→    const grad = ctx.createRadialGradient(screenX, groundY, radiusX * 0.15, screenX, groundY, radiusX);
   377→    grad.addColorStop(0, 'rgba(56, 189, 248, 0.14)');
   378→    grad.addColorStop(0.7, 'rgba(56, 189, 248, 0.05)');
   379→    grad.addColorStop(1, 'rgba(56, 189, 248, 0)');
   380→    ctx.fillStyle = grad;
   381→    ctx.fill();
   382→
   383→    // Calligraphy dash boundary
   384→    ctx.strokeStyle = 'rgba(250, 204, 21, 0.35)';
   385→    ctx.lineWidth = 1.8;
   386→    ctx.setLineDash([8, 12]);
   387→    ctx.lineDashOffset = -gameTime * 25;
   388→    ctx.beginPath();
   389→    ctx.ellipse(screenX, groundY, radiusX, radiusY, 0, 0, Math.PI * 2);
   390→    ctx.stroke();
   391→
   392→    ctx.restore();
   393→  }

   395→  // Draw Ink Lotus Trap
   396→  private drawInkLotus(
   397→    trap: InkLotusTrap,
   398→    cameraX: number,
   399→    laneMinY: number,
   400→    laneMaxY: number,
   401→    gameTime: number
   402→  ) {
   403→    const screenX = trap.x - cameraX;
   404→    if (screenX < -100 || screenX > this.width + 100) return;
   405→
   406→    const ctx = this.ctx;
   407→    const scale = this.getDepthScale(trap.y, laneMinY, laneMaxY);
   408→    const rx = trap.radius * scale;
   409→    const ry = rx * 0.4;
   410→
   411→    ctx.save();
   412→    ctx.translate(screenX, trap.y);
   413→
   414→    // Glowing lotus circle
   415→    ctx.strokeStyle = trap.isBloomed ? 'rgba(52, 211, 153, 0.8)' : 'rgba(56, 189, 248, 0.6)';
   416→    ctx.lineWidth = 2;
   417→    ctx.beginPath();
   418→    ctx.ellipse(0, 0, rx, ry, 0, 0, Math.PI * 2);
   419→    ctx.stroke();
   420→
   421→    // Blossoming Petals
   422→    const numPetals = 8;
   423→    const spin = gameTime * 1.5;
   424→    for (let i = 0; i < numPetals; i++) {
   425→      const angle = (i / numPetals) * Math.PI * 2 + spin;
   426→      const px = Math.cos(angle) * (rx * 0.6);
   427→      const py = Math.sin(angle) * (ry * 0.6);
   428→
   429→      ctx.fillStyle = trap.isBloomed ? '#34d399' : '#38bdf8';
   430→      ctx.beginPath();
   431→      ctx.ellipse(px, py, 6 * scale, 3 * scale, angle, 0, Math.PI * 2);
   432→      ctx.fill();
   433→    }
   434→
   435→    ctx.restore();
   436→  }

   438→  // Enhanced Multi-Joint Skeletal Hero Animation
   439→  private drawSkeletalHero(
   440→    hero: HeroInstance,
   441→    cameraX: number,
   442→    laneMinY: number,
   443→    laneMaxY: number,
   444→    gameTime: number
   445→  ) {
   446→    const screenX = hero.x - cameraX;
   447→    if (screenX < -120 || screenX > this.width + 120) return;
   448→
   449→    const ctx = this.ctx;
   450→    const groundY = hero.y;
   451→    const altitudeZ = hero.altitudeZ || 0;
   452→    const heroY = groundY - altitudeZ;
   453→    const scale = this.getDepthScale(hero.y, laneMinY, laneMaxY);
   454→    const facing = hero.facing;
   455→
   456→    // 1. Draw 2.5D Drop Shadow on ground
   457→    this.drawDropShadow(screenX, groundY, 24 * scale, altitudeZ);
   458→
   459→    ctx.save();
   460→    ctx.translate(screenX, heroY);
   461→    ctx.scale(facing * scale, scale);
   462→
   463→    // Flinch Recoil if taking damage
   464→    if (hero.flinchTimer > 0) {
   465→      ctx.rotate(-0.15);
   466→      ctx.translate(-5, 0);
   467→    }
   468→
   469→    // Skeletal Gait Calculation
   470→    const isMoving = hero.state === 'WALK';
   471→    const walkFreq = 12;
   472→    const gaitPhase = isMoving ? gameTime * walkFreq : 0;
   473→
   474→    // Pelvis & Body Bobbing
   475→    const hipBob = isMoving ? Math.abs(Math.sin(gaitPhase)) * 4 : Math.sin(gameTime * 2) * 1.5;
   476→    const hipY = -28 + hipBob;
   477→
   478→    // 2. Billowing Multi-Segment Silk Cape (Fluid physics)
   479→    this.drawMultiSegmentCape(ctx, hipY - 20, isMoving, facing, hero.state, gameTime);
   480→
   481→    // 3. Multi-Joint Legs (Thigh -> Knee -> Shin -> Foot)
   482→    ctx.strokeStyle = '#09090b';
   483→    ctx.lineWidth = 4;
   484→    ctx.lineCap = 'round';
   485→    ctx.lineJoin = 'round';
   486→
   487→    const leg1Swing = isMoving ? Math.sin(gaitPhase) * 0.7 : 0.05;
   488→    const leg2Swing = isMoving ? -Math.sin(gaitPhase) * 0.7 : -0.05;
   489→
   490→    // Left Leg (Back)
   491→    this.drawLimbBone(ctx, -2, hipY, leg1Swing, 14, 14);
   492→
   493→    // Right Leg (Front)
   494→    this.drawLimbBone(ctx, 2, hipY, leg2Swing, 14, 14);
   495→
   496→    // 4. Torso & Robes (Spine bone)
   497→    const spineLean = isMoving ? 0.12 : 0;
   498→    const shoulderX = Math.sin(spineLean) * 22;
   499→    const shoulderY = hipY - 22;
   500→
   501→    ctx.beginPath();
   502→    ctx.moveTo(0, hipY);
   503→    ctx.lineTo(shoulderX, shoulderY);
   504→    ctx.stroke();
   505→
   506→    // Robe Sash / Belt in jade turquoise
   507→    ctx.fillStyle = '#06b6d4';
   508→    ctx.fillRect(-6, hipY - 4, 12, 4);
   509→
   510→    // 5. Head, Neck & Topknot
   511→    const headX = shoulderX + 2;
   512→    const headY = shoulderY - 14;
   513→
   514→    ctx.fillStyle = '#09090b';
   515→    ctx.beginPath();
   516→    ctx.arc(headX, headY, 8.5, 0, Math.PI * 2);
   517→    ctx.fill();
   518→
   519→    // Topknot & Flowing Ribbon
   520→    ctx.beginPath();
   521→    ctx.arc(headX, headY - 10, 4, 0, Math.PI * 2);
   522→    ctx.fill();
   523→
   524→    ctx.strokeStyle = '#06b6d4';
   525→    ctx.lineWidth = 2.2;
   526→    const ribbonWave = Math.sin(gameTime * 6) * 6;
   527→    ctx.beginPath();
   528→    ctx.moveTo(headX, headY - 10);
   529→    ctx.quadraticCurveTo(headX - 12 + ribbonWave * 0.5, headY - 12, headX - 22 + ribbonWave, headY - 6);
   530→    ctx.stroke();
   531→
   532→    // 6. Multi-Joint Arms & Luminous Brush-Sword
   533→    ctx.strokeStyle = '#09090b';
   534→    ctx.lineWidth = 4;
   535→

   585→  // Draw 2-joint limb bone (thigh/shin or arm/forearm)
   586→  private drawLimbBone(
   587→    ctx: CanvasRenderingContext2D,
   588→    startX: number,
   589→    startY: number,
   590→    rootAngle: number,
   591→    len1: number,
   592→    len2: number
   593→  ) {
   594→    const kneeX = startX + Math.sin(rootAngle) * len1;
   595→    const kneeY = startY + Math.cos(rootAngle) * len1;
   596→
   597→    // Joint 2 angle bends naturally forward/backward
   598→    const bendAngle = rootAngle + 0.35;
   599→    const footX = kneeX + Math.sin(bendAngle) * len2;
   600→    const footY = kneeY + Math.cos(bendAngle) * len2;
   601→
   602→    ctx.beginPath();
   603→    ctx.moveTo(startX, startY);
   604→    ctx.lineTo(kneeX, kneeY);
   605→    ctx.lineTo(footX, footY);
   606→    ctx.stroke();

  1139→  private drawProjectile(
  1140→    proj: Projectile,
  1141→    cameraX: number,
  1142→    laneMinY: number,
  1143→    laneMaxY: number
  1144→  ) {
  1145→    const screenX = proj.x - cameraX;

  1147→

  1150→    const screenY = proj.y - altitudeZ;

  1152→
  1153→    ctx.save();
  1154→    ctx.translate(screenX, screenY);
  1155→    ctx.scale(scale, scale);
  1156→
  1157→    const angle = Math.atan2(proj.vy, proj.vx);
  1158→    ctx.rotate(angle);
  1159→
  1160→    if (proj.type === 'ARROW') {
  1161→      ctx.strokeStyle = proj.isEnemy ? '#ef4444' : '#22c55e';
  1162→      ctx.lineWidth = 2;

  1164→      ctx.moveTo(-16, 0);
  1165→      ctx.lineTo(8, 0);

  1171→      ctx.lineTo(14, 0);
  1172→      ctx.lineTo(8, 3);
  1173→      ctx.closePath();
  1174→      ctx.fill();
  1175→    } else if (proj.type === 'QUILL') {
  1176→      const quillGrad = ctx.createLinearGradient(-24, 0, 12, 0);
  1177→      quillGrad.addColorStop(0, 'rgba(250, 204, 21, 0)');

  1189→      ctx.shadowBlur = 8;
  1190→      ctx.beginPath();

  1196→      ctx.beginPath();
  1197→      ctx.arc(0, 0, 16, -Math.PI * 0.45, Math.PI * 0.45);
